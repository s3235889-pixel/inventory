// Google Apps Script (GAS) 後端邏輯 function doGet(e) { const action = e.parameter.action; const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = sheet.getDataRange().getValues(); const headers = data[0]; const rows = data.slice(1); // 1. 查詢/讀取全部資料 if (!action || action === 'read') { const result = rows.map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); return responseJSON({ status: 'success', data: result }); } } function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const body = JSON.parse(e.postData.contents); const action = body.action; // 2. 新增資料 if (action === 'create') { const newId = new Date().getTime().toString(); // 用時間戳當簡單 ID const now = new Date().toLocaleString('zh-TW', { timeZone: 'Asia/Taipei' }); // 已修正：使用 body.itemName (大寫 N) sheet.appendRow([newId, body.itemName, body.quantity, now]); return responseJSON({ status: 'success', message: '新增成功' }); } // 3. 修改資料 if (action === 'update') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0].toString() === body.id.toString()) { const now = new Date().toLocaleString('zh-TW', { timeZone: 'Asia/Taipei' }); // 已修正：更新時使用 body.itemName (大寫 N) sheet.getRange(i + 1, 2).setValue(body.itemName); sheet.getRange(i + 1, 3).setValue(body.quantity); sheet.getRange(i + 1, 4).setValue(now); return responseJSON({ status: 'success', message: '修改成功' }); } } return responseJSON({ status: 'error', message: '找不到該筆資料' }); } // 4. 刪除資料 if (action === 'delete') { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0].toString() === body.id.toString()) { sheet.deleteRow(i + 1); return responseJSON({ status: 'success', message: '刪除成功' }); } } return responseJSON({ status: 'error', message: '找不到該筆資料' }); } } // 輔助函式：回傳 JSON 格式 function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
